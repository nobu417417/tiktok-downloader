// Bilibili用処理（エラーハンドリングの強化例） async function handleBilibili(rawUrl) { const statusDiv = document.getElementById('status'); const parsed = parseBilibiliInfo(rawUrl); const bvid = parsed.videoId; if (!bvid) { lastErrorMessage = '[Bilibili Error] BV号が見つかりません。'; return false; } statusDiv.innerText = '解析を試行中...'; // 1. 基本情報の取得（UP主ページの特定） try { const viewApi = `https://api.bilibili.com/x/web-interface/view?bvid=${bvid}`; const proxyRes = await fetch(`https://api.allorigins.win/get?url=${encodeURIComponent(viewApi)}`); if (proxyRes.ok) { const wrapper = await proxyRes.json(); const bData = JSON.parse(wrapper.contents); if (bData.code === 0 && bData.data?.owner?.mid) { showUserLink('bilibili', bData.data.owner.mid); } } } catch (e) { console.warn('UP主情報の取得に失敗:', e); } // 2. 自動取得が失敗した場合の案内 lastErrorMessage = 'API取得制限により、直リンクの自動解析が拒否されました。'; // ユーザーに手動操作や代替方法を案内するUI表示 statusDiv.innerHTML = ` 自動取得に失敗しました。
Bilibiliのセキュリティ制限のため、動画ページから直接確認してください。 `; return false; }
